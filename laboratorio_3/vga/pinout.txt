set_location_assignment PIN_AA17 -to VGA_G[2]

# VGA Blue (4 bits)

set_location_assignment PIN_AB18 -to VGA_B[1]
set_location_assignment PIN_AA19 -to VGA_B[2]
set_location_assignment PIN_AB19 -to VGA_B[3]
